function saveSelectedPlan() { const selectedPathways = Array.from(document.querySelectorAll('input[name="careerChoice"]:checked')).map(cb => cb.parentElement.innerText.trim()); if (selectedPathways.length === 0) { alert("Please select at least one career pathway first."); return; } // Gather chosen skills const selectedSkills = Array.from(document.querySelectorAll('#skillsChecklistContainer input[type="checkbox"]:checked')).map(cb => cb.parentElement.innerText.trim()); // Research domain (if applicable) const isResearch = document.querySelector('input[name="careerChoice"][value="research"]').checked; const researchDomain = isResearch ? document.getElementById("researchFieldSelect").selectedOptions[0].text : "N/A"; const facultyMentor = isResearch ? document.getElementById("facultyMatchText").innerText : "N/A"; const resScore = document.getElementById("resScoreDisplay").innerText; const indScore = document.getElementById("indScoreDisplay").innerText; // 1. Save to Browser Storage const idpPayload = { date: new Date().toLocaleDateString(), researchScore: resScore, industryScore: indScore, pathways: selectedPathways, researchDomain: researchDomain, mentorAdvisory: facultyMentor, skills: selectedSkills }; localStorage.setItem("careerCompassIDP", JSON.stringify(idpPayload)); // 2. Generate and Download Formal IDP Dossier let dossierText = "====================================================\n"; dossierText += " CAREER-COMPASS: INDIVIDUAL DEVELOPMENT PLAN (IDP)\n"; dossierText += " School of Biosciences & Biotechnology\n"; dossierText += "====================================================\n\n"; dossierText += `Generated Date: ${idpPayload.date}\n`; dossierText += `Diagnostic Fit: Research: ${resScore} | Industry: ${indScore}\n\n`; dossierText += "--- TARGET CAREER PATHWAYS ---\n"; selectedPathways.forEach(p => dossierText += `• ${p}\n`); if (isResearch) { dossierText += "\n--- RESEARCH SPECIALIZATION & MENTOR ---\n"; dossierText += `Domain: ${researchDomain}\n`; dossierText += `Advisor: ${facultyMentor}\n`; } dossierText += "\n--- SELECTED SKILL MODULES TO DEVELOP ---\n"; selectedSkills.forEach(s => dossierText += `[ ] ${s}\n`); dossierText += "\n====================================================\n"; dossierText += "Next Steps: Submit this dossier to the Placement Coordinator\n"; dossierText += "or bring to Faculty Office Hours for signature.\n"; dossierText += "====================================================\n"; const blob = new Blob([dossierText], { type: "text/plain" }); const url = window.URL.createObjectURL(blob); const a = document.createElement("a"); a.href = url; a.download = "My_Career_Compass_IDP.txt"; a.click(); window.URL.revokeObjectURL(url); alert(`Your Individual Development Plan has been saved in your browser and downloaded as 'My_Career_Compass_IDP.txt'.\n\nYou can share this directly with the Placement Coordinator or Research Committee.`); }
